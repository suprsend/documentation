    ### Step 3: Implementation

    ```javascript
    const { Suprsend, WorkflowTriggerRequest } = require('@suprsend/node-sdk');

    const suprClient = new Suprsend(
      process.env.SUPERSEND_WORKSPACE_KEY,
      process.env.SUPERSEND_WORKSPACE_SECRET
    );

    /**
     * Create shared channel with user and invite them
     */
    async function createSharedChannel(userEmail, channelName) {
      try {
        // Step 1: Create channel
        const channelResponse = await fetch('https://slack.com/api/conversations.create', {
          method: 'POST',
          headers: {
            'Authorization': `Bearer ${process.env.SLACK_BOT_TOKEN}`,
            'Content-Type': 'application/json',
          },
          body: JSON.stringify({
            name: channelName,
            is_private: false,
          }),
        });

        const channelData = await channelResponse.json();
        
        if (!channelData.ok) {
          throw new Error(`Failed to create channel: ${channelData.error}`);
        }

        // Step 2: Look up user ID by email
        const userLookup = await fetch(`https://slack.com/api/users.lookupByEmail?email=${encodeURIComponent(userEmail)}`, {
          method: 'GET',
          headers: {
            'Authorization': `Bearer ${process.env.SLACK_BOT_TOKEN}`,
          },
        });
        const userData = await userLookup.json();
        
        if (!userData.ok) {
          throw new Error(`User lookup failed: ${userData.error}`);
        }

        // Step 3: Invite user to channel using user ID
        const inviteResponse = await fetch('https://slack.com/api/conversations.invite', {
          method: 'POST',
          headers: {
            'Authorization': `Bearer ${process.env.SLACK_BOT_TOKEN}`,
            'Content-Type': 'application/json',
          },
          body: JSON.stringify({
            channel: channelData.channel.id,
            users: userData.user.id, // Use user ID, not email
          }),
        });

        const inviteData = await inviteResponse.json();
        
        if (!inviteData.ok) {
          console.warn(`Failed to invite user: ${inviteData.error}`);
        }

        return {
          channelId: channelData.channel.id,
          channelName: channelData.channel.name,
          success: true,
          inviteSuccess: inviteData.ok
        };

      } catch (error) {
        console.error('Error creating shared channel:', error);
        throw error;
      }
    }

    /**
     * Create SuprSend user profile for channel communication
     */
    async function createChannelUserProfile(userInfo, channelId) {
      try {
        const distinctId = `channel_${userInfo.email.replace('@', '_at_')}`;
        const user = suprClient.user.get_instance(distinctId);
        
        // Add Slack channel information
        user.add_slack({
          channel_id: channelId,
          access_token: process.env.SLACK_BOT_TOKEN,
        });
        
        // Set user properties
        user.set_once('name', userInfo.name);
        user.set_once('email', userInfo.email);
        user.set_once('channel_type', 'company_channel');
        user.set_once('joined_at', new Date().toISOString());
        
        const response = await user.save();
        return { distinct_id: distinctId, ...response };
        
      } catch (error) {
        console.error('Error creating channel user profile:', error);
        throw error;
      }
    }

    /**
     * Send welcome message to company channel
     */
    async function sendChannelWelcomeMessage(distinctId, userInfo) {
      try {
        const body = {
          workflow: 'company_channel_welcome',
          recipients: [
            {
              distinct_id: distinctId,
              name: userInfo.name,
            }
          ],
          data: {
            user_name: userInfo.name,
            user_email: userInfo.email,
            channel_name: userInfo.channelName,
            welcome_message: `Welcome ${userInfo.name}! We're excited to have you in our community.`,
            getting_started_url: 'https://yourdomain.com/getting-started',
            support_channel: '#support',
            community_guidelines: 'https://yourdomain.com/guidelines',
          }
        };
        
        const workflowTrigger = new WorkflowTriggerRequest(body);
        const response = await suprClient.workflows.trigger(workflowTrigger);
        return response;
        
      } catch (error) {
        console.error('Error sending channel welcome message:', error);
        throw error;
      }
    }

    /**
     * Complete company channel integration flow
     */
    async function handleCompanyChannelIntegration(userInfo) {
      try {
        // Step 1: Create shared channel
        const channelResult = await createSharedChannel(
          userInfo.email, 
          `community-${userInfo.name.toLowerCase().replace(/\s+/g, '-')}`
        );
        
        // Step 2: Create SuprSend user profile
        const userProfile = await createChannelUserProfile(userInfo, channelResult.channelId);
        
        // Step 3: Send welcome message
        await sendChannelWelcomeMessage(userProfile.distinct_id, {
          ...userInfo,
          channelName: channelResult.channelName
        });
        
        return {
          channelId: channelResult.channelId,
          channelName: channelResult.channelName,
          distinctId: userProfile.distinct_id,
          success: true
        };
        
      } catch (error) {
        console.error('Company channel integration failed:', error);
        throw error;
      }
    }

    // Usage example
    const userInfo = {
      name: 'John Doe',
      email: 'john@example.com'
    };

    handleCompanyChannelIntegration(userInfo)
      .then(result => console.log('Integration result:', result))
      .catch(error => console.error('Integration failed:', error));
    ```

    ### Step 2: Implementation

    ```javascript
    /**
     * Generate Slack OAuth authorization URL for user workspace
     */
    function generateUserWorkspaceAuthUrl(userId) {
      const params = new URLSearchParams({
        client_id: process.env.SLACK_CLIENT_ID,
        scope: 'channels:read,chat:write,im:write,users:read,users:read.email',
        redirect_uri: process.env.SLACK_REDIRECT_URI,
        state: userId,
      });
      
      return `https://slack.com/oauth/v2/authorize?${params.toString()}`;
    }

    /**
     * Exchange authorization code for access token
     */
    async function exchangeCodeForToken(code) {
      try {
        const response = await fetch('https://slack.com/api/oauth.v2.access', {
          method: 'POST',
          headers: {
            'Content-Type': 'application/x-www-form-urlencoded',
          },
          body: new URLSearchParams({
            client_id: process.env.SLACK_CLIENT_ID,
            client_secret: process.env.SLACK_CLIENT_SECRET,
            code: code,
            redirect_uri: process.env.SLACK_REDIRECT_URI,
          }),
        });

        const data = await response.json();
        
        if (!data.ok) {
          throw new Error(`Token exchange failed: ${data.error}`);
        }
        
        return data;
      } catch (error) {
        console.error('Error exchanging code for token:', error);
        throw error;
      }
    }

    /**
     * Fetch user's accessible channels
     */
    async function fetchUserChannels(accessToken) {
      try {
        const response = await fetch('https://slack.com/api/conversations.list', {
          method: 'GET',
          headers: {
            'Authorization': `Bearer ${accessToken}`,
          },
        });
        
        const data = await response.json();
        
        if (!data.ok) {
          throw new Error(`Failed to fetch channels: ${data.error}`);
        }
        
        const channels = data.channels
          .filter(channel => !channel.is_archived)
          .map(channel => ({
            id: channel.id,
            name: channel.name,
            is_private: channel.is_private,
            is_member: channel.is_member,
            num_members: channel.num_members,
          }));
        
        return channels;
        
      } catch (error) {
        console.error('Error fetching user channels:', error);
        throw error;
      }
    }

    /**
     * Create SuprSend user profile for workspace integration
     */
    async function createUserWorkspaceProfile(userId, tokenData, channels) {
      try {
        const distinctId = `workspace_${userId}`;
        const user = suprClient.user.get_instance(distinctId);
        
        // Add Slack workspace information
        user.add_slack({
          user_id: tokenData.authed_user.id,
          access_token: tokenData.access_token,
          workspace_id: tokenData.team.id,
          workspace_name: tokenData.team.name,
        });
        
        // Set user properties
        user.set_once('name', tokenData.authed_user.name);
        user.set_once('email', tokenData.authed_user.email);
        user.set_once('workspace_type', 'user_workspace');
        user.set_once('channels_count', channels.length);
        user.set_once('connected_at', new Date().toISOString());
        
        // Store channels for later use
        user.set_once('available_channels', channels);
        
        const response = await user.save();
        return { distinct_id: distinctId, channels, ...response };
        
      } catch (error) {
        console.error('Error creating user workspace profile:', error);
        throw error;
      }
    }

    /**
     * Send notification to user's Slack workspace
     */
    async function sendUserWorkspaceNotification(userId, notificationData, channelId = null) {
      try {
        const distinctId = `workspace_${userId}`;
        
        const body = {
          workflow: 'user_workspace_notification',
          recipients: [
            {
              distinct_id: distinctId,
              name: notificationData.userName,
            }
          ],
          data: {
            ...notificationData,
            channel_id: channelId,
            notification_type: channelId ? 'channel' : 'dm',
            timestamp: new Date().toISOString(),
          }
        };
        
        const workflowTrigger = new WorkflowTriggerRequest(body);
        const response = await suprClient.workflows.trigger(workflowTrigger);
        return response;
        
      } catch (error) {
        console.error('Error sending user workspace notification:', error);
        throw error;
      }
    }

    // Express routes
    app.get('/slack/connect/:userId', (req, res) => {
      const { userId } = req.params;
      const authUrl = generateUserWorkspaceAuthUrl(userId);
      res.redirect(authUrl);
    });

    app.get('/slack/callback', async (req, res) => {
      const { code, state: userId, error } = req.query;
      
      if (error) {
        console.error('Slack OAuth error:', error);
        return res.redirect(`/slack/error?error=${encodeURIComponent(error)}`);
      }
      
      try {
        const tokenData = await exchangeCodeForToken(code);
        const channels = await fetchUserChannels(tokenData.access_token);
        const userProfile = await createUserWorkspaceProfile(userId, tokenData, channels);
        
        res.redirect(`/slack/success?userId=${userId}&channels=${channels.length}`);
        
      } catch (error) {
        console.error('OAuth callback error:', error);
        res.redirect(`/slack/error?message=${encodeURIComponent(error.message)}`);
      }
    });

    // Usage examples
    // Send DM
    sendUserWorkspaceNotification('user123', {
      userName: 'John Doe',
      alertType: 'System Update',
      message: 'Your account has been upgraded to Pro plan',
      actionUrl: 'https://app.company.com/billing',
    })
    .then(result => console.log('DM sent:', result))
    .catch(error => console.error('DM failed:', error));

    // Send to specific channel
    sendUserWorkspaceNotification('user123', {
      userName: 'John Doe',
      alertType: 'Query Alert',
      queryName: 'Daily Sales Report',
      message: 'Your query returned 0 results',
      dashboardUrl: 'https://metabase.company.com/dashboard/123',
    }, 'C1234567890')
    .then(result => console.log('Channel message sent:', result))
    .catch(error => console.error('Channel message failed:', error));
    ```

    /**
     * Set up webhook channel for notifications
     */
    async function setupWebhookChannel(channelName, webhookUrl, purpose = 'notifications') {
      try {
        const distinctId = `webhook_${channelName}`;
        const user = suprClient.user.get_instance(distinctId);
        
        // Add webhook information
        user.add_slack({
          incoming_webhook: {
            url: webhookUrl
          }
        });
        
        // Set channel properties
        user.set_once('name', `#${channelName} (Webhook)`);
        user.set_once('channel_type', 'webhook');
        user.set_once('purpose', purpose);
        user.set_once('created_at', new Date().toISOString());
        
        const response = await user.save();
        return response;
        
      } catch (error) {
        console.error(`Error setting up webhook channel ${channelName}:`, error);
        throw error;
      }
    }

    /**
     * Send notification via webhook
     */
    async function sendWebhookNotification(channelName, notificationData) {
      try {
        const distinctId = `webhook_${channelName}`;
        
        const body = {
          workflow: 'webhook_notification',
          recipients: [
            {
              distinct_id: distinctId,
              name: `#${channelName}`,
            }
          ],
          data: {
            ...notificationData,
            channel_name: channelName,
            timestamp: new Date().toISOString(),
          }
        };
        
        const workflowTrigger = new WorkflowTriggerRequest(body);
        const response = await suprClient.workflows.trigger(workflowTrigger);
        return response;
        
      } catch (error) {
        console.error(`Error sending webhook notification to ${channelName}:`, error);
        throw error;
      }
    }

    // Initialize different webhook channels
    async function initializeWebhookChannels() {
      try {
        await setupWebhookChannel('alerts', process.env.SLACK_WEBHOOK_ALERTS, 'system_alerts');
        await setupWebhookChannel('deployments', process.env.SLACK_WEBHOOK_DEPLOYMENTS, 'deployment_notifications');
        await setupWebhookChannel('support', process.env.SLACK_WEBHOOK_SUPPORT, 'customer_support');
      } catch (error) {
        console.error('Error initializing webhook channels:', error);
        throw error;

    }

    // Call this during application startup

### Security Considerations

<Warning>
  **Token Storage**: SuprSend encrypts tokens at rest, but remember:
  - Users can revoke tokens at any time
  - Implement error handling for revoked tokens
  - Consider token refresh strategies for long-lived integrations
</Warning>

<Info>
  **Best Practices**:
  - Always validate tokens before making API calls
  - Implement proper error handling for all API responses
  - Use environment variables for sensitive data
  - Test webhook URLs in a staging environment first
  - Monitor API rate limits and implement backoff strategies
</Info>

### Troubleshooting Common Issues

<AccordionGroup>
  <Accordion title="OAuth Flow Not Working">
    **Symptoms**: Users can't complete OAuth authorization
    
    **Common Causes**:
    - Incorrect redirect URI
    - Missing required scopes
    - Invalid client ID/secret
    
    **Solutions**:
    1. Verify redirect URI matches exactly in Slack app settings
    2. Check that all required scopes are added
    3. Regenerate client secret if needed
    4. Test OAuth flow in Slack's OAuth tester
  </Accordion>
  
  <Accordion title="Messages Not Sending">
    **Symptoms**: SuprSend workflows trigger but messages don't appear in Slack
    
    **Common Causes**:
    - Invalid channel ID
    - Bot not in channel
    - Insufficient permissions
    
    **Solutions**:
    1. Verify channel ID is correct
    2. Ensure bot is added to the channel
    3. Check bot has required scopes
    4. Test with `auth.test` endpoint
  </Accordion>
  
  <Accordion title="User Invitation Fails">
    **Symptoms**: `conversations.invite` returns error
    
    **Common Causes**:
    - Using email instead of user ID
    - User already in channel
    - User not in workspace
    
    **Solutions**:
    1. Use `users.lookupByEmail` to get user ID first
    2. Check if user is already a member
    3. Verify user exists in workspace
    4. Handle `already_in_channel` error gracefully
  </Accordion>
  
  <Accordion title="Webhook Not Receiving Messages">
    **Symptoms**: Webhook URL configured but no messages received
    
    **Common Causes**:
    - Invalid webhook URL
    - Channel doesn't exist
    - Webhook disabled
    
    **Solutions**:
    1. Test webhook URL with curl
    2. Verify channel exists and is active
    3. Check webhook is enabled in Slack app
    4. Ensure correct JSON payload format
  </Accordion>
</AccordionGroup>
