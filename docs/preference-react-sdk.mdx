The `@suprsend/react` SDK provides headless methods to fetch and update user notification preferences. You build the UI — the SDK handles the API calls, optimistic updates, and real-time sync.

## Prerequisites

- [Install `@suprsend/react` and set up `SuprSendProvider`](/docs/react-installation)
- [Configure notification categories](/docs/user-preferences#create-notification-category) on the SuprSend dashboard

## Integration steps

<Steps>
  <Step title="Set up SuprSendProvider">
    Wrap your app in [`SuprSendProvider`](/docs/react-installation#set-up-the-provider) to create the SuprSend client and authenticate the user.
  </Step>

  <Step title="Access preference methods">
    Call `useSuprSendClient()` inside your component to get the client instance, which exposes all preference methods under `client.user.preferences`.
  </Step>

  <Step title="Fetch and render preferences">
    Call `getPreferences()` to fetch the user's preference data, then render it. Listen to `preferences_updated` and `preferences_error` events to keep the UI in sync.
  </Step>
</Steps>

## Understanding the data structure

Preferences contain two sections — refer to the [JavaScript Preferences guide](/docs/js-preferences#understanding-preference-structure) for the complete data structure documentation:

1. **Category-level preferences** — Sections with categories and per-channel toggles
2. **Overall channel preferences** — Global channel-level settings (All vs Required)

```typescript
interface PreferenceData {
  sections: Section[] | null;
  channel_preferences: ChannelPreference[] | null;
}
```

## Available methods

| Method | Description |
|--------|-------------|
| `getPreferences(args?)` | Fetch the full preference tree. Accepts optional `tenantId` and `tags`. |
| `updateCategoryPreference(category, preference)` | Opt-in or opt-out of a category. |
| `updateChannelPreferenceInCategory(channel, preference, category)` | Toggle a specific channel within a category. |
| `updateOverallChannelPreference(channel, preference)` | Set a channel to "All" or "Required". |

All update methods are **optimistic** — the UI updates immediately and the API call happens in the background with a 1-second debounce. Listen to event listeners to handle success and error states.

### Event listeners

```typescript
suprSendClient.emitter.on("preferences_updated", (data: ApiResponse) => {
  // Update your UI with data.body
});

suprSendClient.emitter.on("preferences_error", (response: ApiResponse) => {
  // Show error toast
});
```

## Complete example

```jsx
import { useState, useEffect } from "react";
import {
  SuprSendProvider,
  ChannelLevelPreferenceOptions,
  PreferenceOptions,
  useSuprSendClient,
  useAuthenticateUser,
} from "@suprsend/react";

export default function App() {
  return (
    <SuprSendProvider
      publicApiKey="YOUR_PUBLIC_API_KEY"
      distinctId="YOUR_DISTINCT_ID"
    >
      <Preferences />
    </SuprSendProvider>
  );
}

function Preferences() {
  const suprSendClient = useSuprSendClient();
  const { authenticatedUser } = useAuthenticateUser();
  const [preferenceData, setPreferenceData] = useState();

  useEffect(() => {
    if (!authenticatedUser) return;

    suprSendClient.user.preferences.getPreferences().then((resp) => {
      if (resp.status !== "error") {
        setPreferenceData({ ...resp.body });
      }
    });

    suprSendClient.emitter.on("preferences_updated", (data) => {
      setPreferenceData({ ...data.body });
    });

    suprSendClient.emitter.on("preferences_error", (response) => {
      console.log("ERROR:", response?.error?.message);
    });
  }, [authenticatedUser]);

  if (!preferenceData) return <p>Loading...</p>;

  return (
    <div style={{ margin: 24 }}>
      <h3 style={{ marginBottom: 24 }}>Notification Preferences</h3>
      <CategoryPreferences
        data={preferenceData}
        onUpdate={setPreferenceData}
      />
      <ChannelPreferences
        data={preferenceData}
        onUpdate={setPreferenceData}
      />
    </div>
  );
}

function CategoryPreferences({ data, onUpdate }) {
  const suprSendClient = useSuprSendClient();

  const handleCategoryToggle = async (category, checked) => {
    const resp = await suprSendClient.user.preferences.updateCategoryPreference(
      category,
      checked ? PreferenceOptions.OPT_IN : PreferenceOptions.OPT_OUT
    );
    if (resp.status !== "error") onUpdate({ ...resp.body });
  };

  const handleChannelToggle = async (channel, category) => {
    if (!channel.is_editable) return;
    const resp =
      await suprSendClient.user.preferences.updateChannelPreferenceInCategory(
        channel.channel,
        channel.preference === PreferenceOptions.OPT_IN
          ? PreferenceOptions.OPT_OUT
          : PreferenceOptions.OPT_IN,
        category
      );
    if (resp.status !== "error") onUpdate({ ...resp.body });
  };

  return data.sections?.map((section, i) => (
    <div key={i} style={{ marginBottom: 24 }}>
      {section.name && (
        <div style={{ backgroundColor: "#FAFBFB", padding: "12px 0", marginBottom: 18 }}>
          <p style={{ fontSize: 18, fontWeight: 500 }}>{section.name}</p>
          <p style={{ color: "#6C727F" }}>{section.description}</p>
        </div>
      )}
      {section.subcategories?.map((sub, j) => (
        <div key={j} style={{ borderBottom: "1px solid #D9D9D9", paddingBottom: 12, marginTop: 18 }}>
          <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center" }}>
            <div>
              <p style={{ fontSize: 16, fontWeight: 600 }}>{sub.name}</p>
              <p style={{ color: "#6C727F", fontSize: 14 }}>{sub.description}</p>
            </div>
            <input
              type="checkbox"
              disabled={!sub.is_editable}
              checked={sub.preference === PreferenceOptions.OPT_IN}
              onChange={(e) => handleCategoryToggle(sub.category, e.target.checked)}
            />
          </div>
          <div style={{ display: "flex", gap: 10, marginTop: 12 }}>
            {sub.channels?.map((ch, k) => (
              <label key={k} style={{ cursor: ch.is_editable ? "pointer" : "not-allowed" }}>
                <input
                  type="checkbox"
                  disabled={!ch.is_editable}
                  checked={ch.preference === PreferenceOptions.OPT_IN}
                  onChange={() => handleChannelToggle(ch, sub.category)}
                />
                {" "}{ch.channel}
              </label>
            ))}
          </div>
        </div>
      ))}
    </div>
  ));
}

function ChannelPreferences({ data, onUpdate }) {
  const suprSendClient = useSuprSendClient();

  const handleChange = async (channel, status) => {
    const resp =
      await suprSendClient.user.preferences.updateOverallChannelPreference(
        channel,
        status
      );
    if (resp.status !== "error") onUpdate({ ...resp.body });
  };

  return (
    <div>
      <div style={{ backgroundColor: "#FAFBFB", padding: "12px 0", marginBottom: 18 }}>
        <p style={{ fontSize: 18, fontWeight: 500 }}>Channel preferences</p>
      </div>
      {data.channel_preferences?.map((ch, i) => (
        <div key={i} style={{ border: "1px solid #D9D9D9", borderRadius: 5, padding: "12px 24px", marginBottom: 24 }}>
          <p style={{ fontSize: 18, fontWeight: 500 }}>{ch.channel}</p>
          <div style={{ marginTop: 12, marginLeft: 24 }}>
            <label style={{ display: "block", marginBottom: 8 }}>
              <input
                type="radio"
                checked={!ch.is_restricted}
                onChange={() => handleChange(ch.channel, ChannelLevelPreferenceOptions.ALL)}
              />
              {" "}All — Allow all notifications except those turned off
            </label>
            <label style={{ display: "block" }}>
              <input
                type="radio"
                checked={ch.is_restricted}
                onChange={() => handleChange(ch.channel, ChannelLevelPreferenceOptions.REQUIRED)}
              />
              {" "}Required — Only important account and security notifications
            </label>
          </div>
        </div>
      ))}
    </div>
  );
}
```
